async function processImages() { const fileInput = document.getElementById('upload'); const files = fileInput.files; if (files.length === 0) { return alert('দয়া করে অন্তত একটি বা একাধিক ছবি সিলেক্ট করুন!'); } const status = document.getElementById('status'); const gallery = document.getElementById('gallery'); const btn = document.getElementById('process-btn'); gallery.innerHTML = ''; btn.disabled = true; const zip = new JSZip(); const targetWidth = 3840; // Standard 4K Width for (let i = 0; i < files.length; i++) { const file = files[i]; status.innerText = `প্রসেসিং চলছে: ${i + 1} / ${files.length} (${file.name})`; const dataUrl = await readFileAsDataURL(file); const img = await loadImage(dataUrl); // Canvas element creation const canvas = document.createElement('canvas'); const ctx = canvas.getContext('2d'); const scaleFactor = targetWidth / img.width; const targetHeight = img.height * scaleFactor; canvas.width = targetWidth; canvas.height = targetHeight; ctx.imageSmoothingEnabled = true; ctx.imageSmoothingQuality = 'high'; ctx.drawImage(img, 0, 0, targetWidth, targetHeight); // অরিজিনাল ফাইল টাইপ (file.type) ব্যবহার করে কনভার্ট করা হচ্ছে const upscaledDataUrl = canvas.toDataURL(file.type || 'image/jpeg', 0.92); const blob = await (await fetch(upscaledDataUrl)).blob(); // অরিজিনাল ফাইলের নাম (file.name) ব্যবহার করা হচ্ছে const originalFileName = file.name; zip.file(originalFileName, blob); // গ্যালারিতে প্রিভিউ দেখানো const card = document.createElement('div'); card.className = 'card'; card.innerHTML = `

${originalFileName}

`; gallery.appendChild(card); } status.innerText = 'জিপ (ZIP) ফাইল তৈরি করা হচ্ছে...'; // Download Zip zip.generateAsync({ type: "blob" }).then(function(content) { saveAs(content, "upscaled_4k_images.zip"); status.innerText = 'সবগুলো ছবি সফলভাবে 4K-তে কনভার্ট হয়ে Zip ডাউনলোড হয়ে গেছে!'; btn.disabled = false; }); }
